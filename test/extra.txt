# Async function with Await statement

АСИНХ Функция ВычислитьАсинхронно()
  Результат = Ждать Сервер.Вычислить();
  Возврат Результат;
КонецФункции

==>

Module(
  FunctionDecl(
    Async,
    Function,
    SubName(Identifier),
    "(", ")",
    BodyStatement(
      Assignment(
        Identifier, "=",
        AwaitExpr(Await, Identifier, CallAccess(".", Identifier, "(", ")"))
      ),
      ";"
    ),
    BodyStatement(ReturnStatement(Return, Identifier), ";"),
    EndFunction
  )
)

# Goto and label

Процедура Т()
  ~Начало:
  Если а = 1 Тогда
    Перейти ~Начало;
  КонецЕсли;
КонецПроцедуры

==>

Module(
  ProcedureDecl(
    Procedure,
    SubName(Identifier),
    "(", ")",
    BodyStatement(
      Label("~", LabelName(Identifier), ":"),
      IfStatement(
        If,
        BinaryExpr(Identifier, CmpOp, Number),
        Then,
        BodyStatement(GotoStatement(Goto, "~", Identifier), ";"),
        EndIf
      )
    ),
    BodyStatement(";"),
    EndProcedure
  )
)

# Ternary operator and New expression

Процедура Т()
  з = ?(а > 0, "плюс", "минус");
  Объект = Новый Структура("Имя, Возраст", "Иван", 30);
КонецПроцедуры

==>

Module(
  ProcedureDecl(
    Procedure,
    SubName(Identifier),
    "(", ")",
    BodyStatement(
      Assignment(
        Identifier, "=",
        TernaryOp("?", "(",
          BinaryExpr(Identifier, CmpOp, Number), ",",
          String, ",",
          String,
          ")"
        )
      ),
      ";"
    ),
    BodyStatement(
      Assignment(
        Identifier, "=",
        NewExpr(New, Identifier, "(", String, ",", String, ",", Number, ")")
      ),
      ";"
    ),
    EndProcedure
  )
)

# AddHandler / RemoveHandler

Процедура Т()
  ДобавитьОбработчик Событие.OnClick, ОбработчикКлика;
  УдалитьОбработчик Событие.OnClick, ОбработчикКлика;
КонецПроцедуры

==>

Module(
  ProcedureDecl(
    Procedure,
    SubName(Identifier),
    "(", ")",
    BodyStatement(
      AddHandlerStatement(
        AddHandler,
        Identifier, PropertyAccess(".", Identifier),
        ",",
        Identifier
      ),
      ";"
    ),
    BodyStatement(
      RemoveHandlerStatement(
        RemoveHandler,
        Identifier, PropertyAccess(".", Identifier),
        ",",
        Identifier
      ),
      ";"
    ),
    EndProcedure
  )
)

# Preprocessor #Если block with Russian keywords (shared with BSL If/Else)

#Если Клиент Тогда
Процедура НаКлиенте()
КонецПроцедуры
#Иначе
Процедура НаСервере()
КонецПроцедуры
#КонецЕсли

==>

Module(
  Preprocessor(PreprocessorIf("#", If, PreprocSymbol(Identifier), Then)),
  ProcedureDecl(Procedure, SubName(Identifier), "(", ")", EndProcedure),
  Preprocessor(PreprocessorElse("#", Else)),
  ProcedureDecl(Procedure, SubName(Identifier), "(", ")", EndProcedure),
  Preprocessor(PreprocessorEndIf("#", EndIf))
)
