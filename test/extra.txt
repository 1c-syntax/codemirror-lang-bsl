# Async function with Await statement

АСИНХ Функция ВычислитьАсинхронно()
  Результат = Ждать Сервер.Вычислить();
  Возврат Результат;
КонецФункции

==>

Module(
  FunctionDecl(
    Async,
    Function,
    SubName(Identifier),
    LParen, RParen,
    BodyStatement(
      Assignment(
        Identifier, Assign,
        AwaitExpr(Await, Identifier, CallAccess(Dot, Identifier, LParen, RParen))
      ),
      Semicolon
    ),
    BodyStatement(ReturnStatement(Return, Identifier), Semicolon),
    EndFunction
  )
)

# Goto and label

Процедура Т()
  ~Начало:
  Если а = 1 Тогда
    Перейти ~Начало;
  КонецЕсли;
КонецПроцедуры

==>

Module(
  ProcedureDecl(
    Procedure,
    SubName(Identifier),
    LParen, RParen,
    BodyStatement(Label(Tilda, LabelName(Identifier), Colon)),
    BodyStatement(
      IfStatement(
        If,
        BinaryExpr(Identifier, CmpOp(Assign), Number),
        Then,
        BodyStatement(GotoStatement(Goto, Tilda, Identifier), Semicolon),
        EndIf
      ),
      Semicolon
    ),
    EndProcedure
  )
)

# Configuration-extension preprocessor #Удаление / #Вставка

#Удаление
  УстаревшийМетод();
#КонецУдаления
#Вставка
  НовыйМетод();
#КонецВставки

==>

Module(
  Preprocessor(PreprocessorDelete(Hash, PreprocDelete)),
  BodyStatement(CallStatement(Identifier, LParen, RParen), Semicolon),
  Preprocessor(PreprocessorEndDelete(Hash, PreprocEndDelete)),
  Preprocessor(PreprocessorInsert(Hash, PreprocInsert)),
  BodyStatement(CallStatement(Identifier, LParen, RParen), Semicolon),
  Preprocessor(PreprocessorEndInsert(Hash, PreprocEndInsert))
)

# Ternary operator and New expression

Процедура Т()
  з = ?(а > 0, "плюс", "минус");
  Объект = Новый Структура("Имя, Возраст", "Иван", 30);
КонецПроцедуры

==>

Module(
  ProcedureDecl(
    Procedure,
    SubName(Identifier),
    LParen, RParen,
    BodyStatement(
      Assignment(
        Identifier, Assign,
        TernaryOp(Question, LParen,
          BinaryExpr(Identifier, CmpOp(Greater), Number), Comma,
          String, Comma,
          String,
          RParen
        )
      ),
      Semicolon
    ),
    BodyStatement(
      Assignment(
        Identifier, Assign,
        NewExpr(New, Identifier, LParen, String, Comma, String, Comma, Number, RParen)
      ),
      Semicolon
    ),
    EndProcedure
  )
)

# AddHandler / RemoveHandler

Процедура Т()
  ДобавитьОбработчик Событие.OnClick, ОбработчикКлика;
  УдалитьОбработчик Событие.OnClick, ОбработчикКлика;
КонецПроцедуры

==>

Module(
  ProcedureDecl(
    Procedure,
    SubName(Identifier),
    LParen, RParen,
    BodyStatement(
      AddHandlerStatement(
        AddHandler,
        Identifier, PropertyAccess(Dot, Identifier),
        Comma,
        Identifier
      ),
      Semicolon
    ),
    BodyStatement(
      RemoveHandlerStatement(
        RemoveHandler,
        Identifier, PropertyAccess(Dot, Identifier),
        Comma,
        Identifier
      ),
      Semicolon
    ),
    EndProcedure
  )
)

# Preprocessor #Если block with Russian keywords (shared with BSL If/Else)

#Если Клиент Тогда
Процедура НаКлиенте()
КонецПроцедуры
#Иначе
Процедура НаСервере()
КонецПроцедуры
#КонецЕсли

==>

Module(
  Preprocessor(PreprocessorIf(Hash, If, PreprocSymbol(Identifier), Then)),
  ProcedureDecl(Procedure, SubName(Identifier), LParen, RParen, EndProcedure),
  Preprocessor(PreprocessorElse(Hash, Else)),
  ProcedureDecl(Procedure, SubName(Identifier), LParen, RParen, EndProcedure),
  Preprocessor(PreprocessorEndIf(Hash, EndIf))
)
