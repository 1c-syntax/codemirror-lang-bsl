Function Hello() Export

==>

Module(
  FunctionDecl(

    EndFunction
  )
)

# Procedure with parameters and Val/default

Процедура Привет(Знач Имя = "мир", Возраст)
КонецПроцедуры

==>

Module(
  ProcedureDecl(
    Procedure,
    SubName(Identifier),
    "(",
    Param(Val, Identifier, "=", DefaultValue(String)),
    ",",
    Param(Identifier),
    ")",
    EndProcedure
  )
)

# Module-level variable declaration

Перем Глобальная, ВтораяГлобальная Экспорт;

==>

Module(
  VarStatement(
    Var,
    VarDeclaration(Identifier),
    ",",
    VarDeclaration(Identifier, Export),
    ";"
  )
)

# If/Elsif/Else

Процедура Т()
  Если а Тогда
    б = 1;
  ИначеЕсли в Тогда
    г = 2;
  Иначе
    д = 3;
  КонецЕсли;
КонецПроцедуры

==>

Module(
  ProcedureDecl(
    Procedure,
    SubName(Identifier),
    "(",
    ")",
    BodyStatement(
      IfStatement(
        If,
        Identifier,
        Then,
        BodyStatement(Assignment(Identifier, "=", Number), ";"),
        ElsifBranch(
          Elsif,
          Identifier,
          Then,
          BodyStatement(Assignment(Identifier, "=", Number), ";")
        ),
        ElseBranch(
          Else,
          BodyStatement(Assignment(Identifier, "=", Number), ";")
        ),
        EndIf
      ),
      ";"
    ),
    EndProcedure
  )
)

Module(

# For-each loop with method call

Процедура Т()
  Для Каждого Элемент Из Коллекция Цикл
    Сообщить(Элемент.Имя);
  КонецЦикла;
КонецПроцедуры

==>

Module(
  ProcedureDecl(
    Procedure,
    SubName(Identifier),
    "(",
    ")",
    BodyStatement(
      ForEachStatement(
        For,
        Each,
        Identifier,
        In,
        Identifier,
        Do,
        BodyStatement(
          CallStatement(
            Identifier,
            "(",
            Identifier,
            PropertyAccess(".", Identifier),
            ")"
          ),
          ";"
        ),
        EndDo
      ),
      ";"
    ),
    EndProcedure
  )
)
