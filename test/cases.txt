# If/Elsif/Else

Процедура Т()
  Если а Тогда
    б = 1;
  ИначеЕсли в Тогда
    г = 2;
  Иначе
    д = 3;
  КонецЕсли;
КонецПроцедуры

==>

Module(
  ProcedureDecl(
    Procedure,
    SubName(Identifier),
    LParen,
    RParen,
    BodyStatement(
      IfStatement(
        If,
        Identifier,
        Then,
        BodyStatement(Assignment(Identifier, Assign, Number), Semicolon),
        ElsifBranch(
          Elsif,
          Identifier,
          Then,
          BodyStatement(Assignment(Identifier, Assign, Number), Semicolon)
        ),
        ElseBranch(
          Else,
          BodyStatement(Assignment(Identifier, Assign, Number), Semicolon)
        ),
        EndIf
      ),
      Semicolon
    ),
    EndProcedure
  )
)

# For-each loop with method call

Процедура Т()
  Для Каждого Элемент Из Коллекция Цикл
    Сообщить(Элемент.Имя);
  КонецЦикла;
КонецПроцедуры

==>

Module(
  ProcedureDecl(
    Procedure,
    SubName(Identifier),
    LParen,
    RParen,
    BodyStatement(
      ForEachStatement(
        For,
        Each,
        Identifier,
        In,
        Identifier,
        Do,
        BodyStatement(
          CallStatement(
            Identifier,
            LParen,
            Identifier,
            PropertyAccess(Dot, Identifier),
            RParen
          ),
          Semicolon
        ),
        EndDo
      ),
      Semicolon
    ),
    EndProcedure
  )
)

# Annotation as annotation parameter value (nested annotation)

&Пластилин(&Деталька("ИмяАппендера"))
Процедура Аппендер()
КонецПроцедуры

==>

Module(
  ProcedureDecl(
    Annotation(
      Ampersand,
      AnnotationName(Identifier),
      AnnotationParams(
        LParen,
        AnnotationParam(
          Annotation(
            Ampersand,
            AnnotationName(Identifier),
            AnnotationParams(LParen, AnnotationParam(String), RParen)
          )
        ),
        RParen
      )
    ),
    Procedure,
    SubName(Identifier),
    LParen,
    RParen,
    EndProcedure
  )
)
